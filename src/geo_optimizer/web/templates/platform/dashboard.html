    function cookieValue(name) {
      return document.cookie.split(';').map(function (part) { return part.trim(); }).reduce(function (found, part) {
        if (found) return found;
        return part.indexOf(name + '=') === 0 ? decodeURIComponent(part.slice(name.length + 1)) : '';
      }, '');
    }

    async function refreshSearchConnections(domain) {
      searchConnectionDomain = domain || '';
      var status = document.getElementById('search-connections-status');
      var list = document.getElementById('search-connections-list');
      list.replaceChildren();
      if (!domain) { status.textContent = 'Verify your website to connect a search account.'; return; }
      try {
        var response = await fetch('/api/v1/search-connections?domain=' + encodeURIComponent(domain), {headers: authHeaders()});
        var data = await readApiResponse(response);
        if (!response.ok) throw new Error(data.detail || 'Could not load connections.');
        var failed = new URLSearchParams(window.location.search).get('search_connection') === 'failed';
        status.textContent = failed ? 'Connection failed. Check that the provider account has access to this verified domain, then try again.' :
          data.connections.length ? 'Connected accounts for ' + domain : 'Connect an account for ' + domain + '.';
        data.connections.forEach(function (connection) {
          var row = document.createElement('div');
          var title = document.createElement('strong');
          title.textContent = connection.provider === 'google' ? 'Google Search Console' : 'Bing Webmaster Tools';
          row.appendChild(title);
          var details = document.createElement('p');
          var rows = connection.result.rows || [];
          var clicks = 0, impressions = 0;
          rows.forEach(function (item) { clicks += Number(item.clicks || item.Clicks || 0); impressions += Number(item.impressions || item.Impressions || 0); });
          details.textContent = connection.status === 'sync_error' ? 'Sync failed. We will retry on the next scheduled run; reconnect if the problem continues.' :
            connection.last_synced_at ? 'Last synced ' + new Date(connection.last_synced_at).toLocaleString() + ': ' + clicks.toLocaleString() + ' clicks, ' + impressions.toLocaleString() + ' impressions. Search traffic, not AI citations.' : 'Connected. The first daily sync is pending.';
          row.appendChild(details);
          var disconnect = document.createElement('button');
          disconnect.type = 'button'; disconnect.textContent = 'Disconnect';
          disconnect.addEventListener('click', async function () {
            var res = await fetch('/api/v1/search-connections/disconnect', {method:'POST', headers:authHeaders(), body:JSON.stringify({domain:domain, provider:connection.provider})});
            if (!res.ok) { status.textContent = 'Could not disconnect the account.'; return; }
            refreshSearchConnections(domain);
          });
          row.appendChild(disconnect); list.appendChild(row);
        });
      } catch (err) { status.textContent = err.message || 'Could not load connections.'; }
    }

    async function readApiResponse(res) {

      } catch (err) {
        return { detail: text || res.statusText || 'Request failed' };
      }
    }

        var summary = document.createElement('summary');
        summary.textContent = 'Review page fixes';
        details.appendChild(summary);
        var recs = []
          .concat(Array.isArray(page.title_issues) ? page.title_issues : [])
          .concat(Array.isArray(page.meta_issues) ? page.meta_issues : [])
          .concat(Array.isArray(page.schema_issues) ? page.schema_issues : [])
          .concat(Array.isArray(page.recommendations) ? page.recommendations : []);
        if (recs.length) {
          var ul = document.createElement('ul');
          recs.slice(0, 8).forEach(function (rec) {
            var item = document.createElement('li');
            item.textContent = rec;
            ul.appendChild(item);
          });
          details.appendChild(ul);
        } else {
          var p = document.createElement('p');
          p.textContent = page.error || 'No priority recommendations returned for this page.';
          details.appendChild(p);
        }
        li.appendChild(head);
        li.appendChild(score);
        li.appendChild(details);
        list.appendChild(li);
      });
    }

    async function refreshSitemapAuditStatus(domain, opts) {
      opts = opts || {};
      var out = document.getElementById('sitemap-audit-out');
      if (!domain) return null;
      var res = await fetch('/api/v1/runs?kind=sitemap_audit', { headers: authHeaders() });
      var data = await readApiResponse(res);
      var run = res.ok && data.runs && data.runs.length ? data.runs[0] : null;
      if (!run) {
        if (!opts.quiet) out.textContent = 'No page audit runs yet.';
        return null;
      }
      var score = run.score === null || run.score === undefined ? '--' : String(run.score) + '/100';
      out.textContent = 'Page audit ' + run.status + ' · ' + score + (run.result_summary && run.result_summary[0] !== '{' ? ' · ' + run.result_summary : '');
      if (run.status === 'completed') {
        await refreshPageAudits(domain);
      }
      return run;
    }

    function scoreTone(score, band, hasError) {
      if (hasError || band === 'failed') return 'failed';
      if (score >= 86 || band === 'excellent') return 'excellent';
      if (score >= 68 || band === 'good') return 'good';
      if (score >= 36 || band === 'foundation') return 'foundation';
      return 'critical';
    }

    function setProductStep(step, state, label) {
      var item = document.querySelector('[data-product-step="' + step + '"]');
      if (!item) return;
      if (state === 'current') {
        document.querySelectorAll('[data-product-step]').forEach(function (other) {
          other.classList.remove('is-current');
        });
      }
      item.classList.remove('is-current', 'is-done', 'is-waiting');
      item.classList.add('is-' + state);
      var status = item.querySelector('em');
      if (status && label) status.textContent = label;
    }

      const res = await fetch('/api/v1/audit/sitemap', {

        body: JSON.stringify({

          sitemap_url: sitemapUrl,
          max_urls: Math.max(1, Math.min(25, maxUrls || 10)),
          run_now: true,
        }),
      });
      const data = await readApiResponse(res);
      if (!res.ok) {
        out.textContent = renderReportText(data.detail || JSON.stringify(data));
        return;
      }
      out.textContent = renderReportText(data.error
        ? data.error
        : (
            data.status === 'queued'
              ? 'Page audit queued for ' + data.sitemap_url + '. A GEOSITION worker will update this list.'
              : 'Average score: ' + data.average_score + '/100 (' + data.average_band + ') · ' + data.audited_urls + ' pages audited.'
          ));
      if (data.storage_warning) out.textContent += '\n' + data.storage_warning;
      renderSitemapPages(data.pages || []);
      if (data.average_score !== undefined) {
        setProductStep('pages', 'done', 'Pages audited');
        setProductStep('benchmark', 'current', 'Benchmark next');
      }
      if (data.status === 'queued') {
        var attempts = 0;
        var timer = window.setInterval(async function () {
          attempts += 1;
          var run = await refreshSitemapAuditStatus(domain, { quiet: true });
          if (!run || run.status === 'completed' || run.status === 'failed' || attempts >= 30) {
            window.clearInterval(timer);
          }
        }, 4000);
      }
